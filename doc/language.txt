#
# Notation
#
# Name: defines a rule;
# stacked indented lines are alternatives;
# [X] is optional, EXCEPT for Array_reference where it is part of the grammar;
# X ... is repetition;
# UPPER_CASE is a lexer token except EMPTY which means no token;
# bare lowercase words are keywords.
#
# Rules are evaluated in the order listed, 1st match wins
#   E.g. "(NAME)+3" is parsed as cast (1st rule wins)
#        "(NAME)(3)" is parsed as cast (1st rule wins)
#
# Composite rule definitions are evaluated from left to right
#
# In recursive evaluation the deeper recursion level is evaluated greedily
#   E.g. in "if(expr1) if(expr2) statement1 else statement2" else binds to if(expr2)
#

# Program flow

Program:
  Sentences

Sentences:
  [ Sentence ...]

Sentence:
  Block
  Qualified_block
  Defer_block
  Line
  For_statement
  While_statement
  If_statement
  Do_statement
  Try_statement
  
Block:
  { Sentences }

# Qualified block: a block with optional static context and/or computation mode
# qualifiers (fixed order: static, finite, saturate|wrap).
#   static   - the body is evaluated in static context (this == NONE)
#   finite   - throws NON_FINITE on the first non-finite result
#   saturate - clamps integer overflow and narrowing
#   wrap     - wraps integer overflow / truncates narrowing
# static (context) and the modes (computation) are orthogonal and combine freely:
#   static finite { ... }
# finite (float strictness) also combines with saturate/wrap (integer mode):
#   finite saturate { ... }  - clamp overflow, throw on non-finite
# saturate and wrap set the same integer-mode knob and may not be combined.
#
# static/finite/wrap/saturate are reserved keywords - mode(expr) and mode{...}
# would otherwise be ambiguous with a function call and a member block. They may
# not be used as local variable/function names, but remain usable as property/
# attribute names, which are always qualified (e.g. x.wrap(...)).
#
# Note: ; terminates a Line; block-statements (Block, Qualified_block, Defer_block,
# if/while/for/do/try) are not Lines and take no ;. A member block is an expression,
# so as a statement it is a Line and takes ; (like any expression whose value is
# discarded): enemy { ... };
Qualified_block:
  static [finite] [saturate] Block
  static [finite] [wrap] Block
  [finite] [saturate] Block
  [finite] [wrap] Block

# Defer block contents will be executed when the current scope is exited
#
# More precisely, they are executed during stack unwinding in LIFO order,
# All local variable declared after the defer block will be cleared first, the ones
# declared before the defer block are accessible during the execution.
# Throwing exceptions is restricted during the execution of defer blocks - exceptions
# may or may not be recorded but the execution is never interrupted.
# Flow-modifying statements (return/break/continue) at surrounding level are not allowed inside defer block
Defer_block:
  defer Block

# Exceptions
#
# try/catch/throw/exception are reserved keywords (like the mode keywords - they may
# not be used as local variable/function names, but remain usable as property/
# attribute names, which are always qualified).
#
# Exception declaration is a compile-time directive: it registers the name(s) in the
# context's exception registry and emits no code. Valid at program scope only
# (compile-time error elsewhere). Names are the sharing mechanism between code units
# in the same context - a repeated declaration with identical shape returns the
# existing registration; a name re-registered with a different shape (category vs
# type, or a different category set) is a compile-time error. A name is either a
# category (a mask bit) or a type (a code), never both.
#   exception MyCategory;                          - declares a category
#   exception MyException : IOException, Network;  - declares a type in categories
# Built-in interpreter exceptions are pre-registered types; built-in category names
# (e.g. Arithmetic) are pre-registered categories.
# Exception names are single NAMEs for now; dotted library names (mylib.Timeout) are
# a planned extension - the registry stores opaque strings, so only the name rules in
# Exception_declaration/Catch_clause and the throw-constructor lookup need to change.
Exception_declaration:
  exception NAME
  exception NAME : NAME [, NAME ...]

# try/catch: first matching clause wins. A category name matches any type registered
# in that category (mask test), a type name matches exactly (code test). The clause
# always binds the caught exception value to NAME (an exception value carries the
# type code, category mask, message and payload). An unfiltered clause is a
# catch-all and must be the last clause. There is no finally keyword - a defer
# block placed first in the try block provides finally semantics.
#
# The try and catch bodies may be plain or qualified blocks:
#   try finite { ... } catch (e : NonFinite) { ... }
# A qualified body is exactly equivalent to nesting the qualified block inside a
# plain block - "try finite B catch ..." means "try { finite B } catch ..." - so
# the qualifiers apply inside their own scope only and catch clauses always run in
# the ambient context and computation mode. Defer_block is not allowed as a body:
# the try region would cover only the defer registration, while the deferred code
# runs at scope exit, outside the region (and defer-time throws are restricted).
Try_statement:
  try Protected_block Catch_clause [Catch_clause ...]

Protected_block:
  Block
  Qualified_block

Catch_clause:
  catch (NAME : NAME [, NAME ...]) Protected_block
  catch (NAME) Protected_block

# throw takes an exception value. Calling a registered exception name as a function
# constructs one - MyException("message" [, payload]) - the name supplies the type
# code and category mask, the arguments fill the message and the optional payload
# value. Re-throwing (throw e;) preserves the original throw site and traceback.
Throw_statement:
  throw Expression

# Declarations

Declaration:
  Declaration_qualifiers Type Single_declaration [, Single_declaration ...]

# Qualifiers:
#   static - per-closure storage: the variable is added to the closure (writable),
#            initialized at closure creation
#   shared - per-lambda storage: the variable lives in the shared datablock of the
#            lambda declaration (all closures of the same lambda share it);
#            the initializer must be a constant expression
#   final  - read-only (cannot be reassigned after initialization)
#   weak   - weak reference (AZWeakReference, AZActiveObject only): reads as null
#            after the referent's shutdown; requires fixed-address storage -
#            valid at program/script scope, compile-time error on function locals
#.  const  - the instance is immutable at least starting from the assignment time.
#            Property and attribute assignments should be errors and optimizer can
#            cache property/attribute access. 
Declaration_qualifiers:
  [static] [final] [weak] [const]
  [shared] [final] [weak] [const]

Single_declaration:
  Pure_declaration
  Declaration_initialization

Pure_declaration:
  NAME

Declaration_initialization:
  Pure_declaration = Expression

Type:
  Type_expression

Cast_expression:

# Cast qualifiers (keywords):
#   exact - the cast throws unless the result is exact (no rounding, no clamping)
#   rounded - the cast allows rounding but throws on clamping
#   (unqualified cast allows both rounding and clamping)
# Reinterpretation (bit casts) is a separate construct, not a cast qualifier

# Primitive types (the only valid cast targets - class/interface conversions use the 'as' operator)
# primitive type names are keywords evaluating to their class, so a parenthesized primitive type
# alone, e.g. (int32), is the class value, not a cast

  int8
  uint8
  int16
  uint16
  int32
  uint32
  int64
  uint64
  float
  double
  complex float
  complex double
  pointer

# The expression form evaluates a single expression in the given mode and
# yields its value. Inside a member block an unqualified mode keyword wins over
# a member of the same name - write this.wrap(...) to call the member.
Mode_expression:

Array_reference:

# Member block (postfix expression operator, same precedence as . () []):
# the head expression is evaluated once, the body is executed in its context
# (this == head value, unqualified names resolve to its properties first, then
# attributes, new names are installed as attributes) and the block yields the
# head value. The head type must be known at compile time.
# A class-valued head gives a class context: classes are pre-built (C code) and
# cannot be modified from scripts, so unqualified names resolve to existing class
# properties or are compile-time errors - no new names can be installed.
# Typical use - configuring a fresh object:
#   Button btn = new Button(parent) {
#     width = 32;
#     onClicked = () => { ... };
#   };
# Postfix { can only follow a complete expression, so it never collides with
# Block ({ at sentence start), Structured_literal ({[...]}) or lambda bodies
# ({ directly after =>).
Member_block:
  Expression Block

RValue:

  this
  Literal

  Function_call

Literal:

Structured_literal:
  {[Expression [, Expression ...]] }

# A type expression evaluates to a class. Currently restricted to a NAME[.NAME...]
# path that must be resolvable to a class at compile time through globals/global
# namespaces (link time - the resolver reads namespace attributes directly; general
# property/attribute access and value folding belong to the optimizer and runtime).
# This keeps function signatures, member contexts and new targets statically known.
# Future extension (not implemented): general type expressions evaluating to a class
# at runtime - e.g. a variable declared with a runtime-resolved type behaves as
# 'any' with a runtime type check at each assignment.
Type_expression:
  NAME [. NAME ...]

Assignment:
  Chained_assignment
  LValue += Expression
  LValue -= Expression
  LValue *= Expression
  LValue /= Expression
  LValue %= Expression
  LValue >>= Expression
  LValue <<= Expression
  LValue &= Expression

# Plain = can be chained: every target except the last one has to be an LValue,
# i.e. a = b = c = 0 assigns 0 to c, b and a (value is evaluated once, targets
# are assigned right to left)
# Note that assignment is a statement, not an expression - a = b is only valid
# where a statement is expected
# Assignment to member/attribute creates member context for a lambda, if chained,
# the last LValue in chain determines the context

Chained_assignment:
  LValue = Expression
  LValue = LValue [= LValue ...] = Expression

# Functions and new

# new takes a type expression (NAME[.NAME...] path) - the class must be resolvable
# at compile time (see Type_expression); it is sugar for Type.new(Arguments)
New:
  new Type Arguments

Arguments:
  ([Expression [, Expression ...]])

# Function definitions are lambdas - [static] (arguments) [return_type] => body
# The body is either a block (explicit return) or a single expression (implicit return)
# Argument types are optional but either all arguments are typed or all are untyped;
# the return type may only be given if the argument list is empty or all arguments
# are typed (for untyped arguments it would be ambiguous with a call of the
# parenthesized argument)
# Without static a lambda created in a member context (a member block, or assigned
# to a member/attribute) captures this - an ad-hoc member function; in a class
# context it captures the class namespace (unqualified names resolve to existing
# class properties) but is not a method - classes cannot be modified from scripts.
# static forces a plain function with exactly the declared signature in any context.
#
# Evaluating a lambda expression creates a new closure every time; captures are
# read-only snapshots. All AZActiveObject captures (including this) are weak -
# closures never extend ActiveObject lifetimes; a dead capture reads as null.
# Non-ActiveObject captures are by-value/strong.
# The optimizer may reuse a closure only if it has no non-constant captures and
# no static variables; closure identity (===) of such lambdas is unspecified.

Lambda:
  [static] (Arguments_definition) => Block
  [static] (Arguments_definition) => Expression
  [static] (Arguments_definition) Type => Block
  [static] (Arguments_definition) Type => Expression

Arguments_definition:
  ([Argument_definition [, Argument_definition ...]])

Argument_definition:
  Type NAME
  NAME

Function_call:
  Expression Arguments
